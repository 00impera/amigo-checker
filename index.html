Generator AMIGO




import random
from pyscript import document, window
from pyjs import to_js

def generate_amigo_draw():
    numbers = list(range(1, 29))
    blue = sorted(random.sample(numbers, 7))
    remaining = [n for n in numbers if n not in blue]
    yellow = sorted(random.sample(remaining, 5))
    return blue, yellow

def amigo_run(event=None):
    blue, yellow = generate_amigo_draw()
    blue_html = " ".join([f"{n}" for n in blue])
    yellow_html = " ".join([f"{n}" for n in yellow])
    result = f"Numere albastre (7): {blue_html}\nNumere galbene (5): {yellow_html}"
    document.getElementById("output").innerHTML = result

document.getElementById("run").addEventListener("click", amigo_run)
