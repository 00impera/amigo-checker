Generator & Verificator AMIGO

Introdu numerele tale (7 numere între 1 și 28):




import random

def generate_amigo_draw():
    numbers = list(range(1, 29))
    blue = random.sample(numbers, 7)
    remaining = [n for n in numbers if n not in blue]
    yellow = random.sample(remaining, 5)
    return sorted(blue), sorted(yellow)

def check_ticket(user_nums, blue, yellow):
    user_set = set(user_nums)
    blue_hits = len(user_set & set(blue))
    yellow_hits = len(user_set & set(yellow))
    return blue_hits, yellow_hits

def amigo_run(user_input):
    try:
        user_nums = [int(x) for x in user_input.split(",")]
        if len(user_nums) != 7:
            return "Trebuie EXACT 7 numere!"
        if any(n < 1 or n > 28 for n in user_nums):
            return "Numerele trebuie să fie între 1 și 28!"
    except:
        return "Format invalid. Exemplu: 3,7,12,14,18,22,27"

    blue, yellow = generate_amigo_draw()
    blue_hits, yellow_hits = check_ticket(user_nums, blue, yellow)

    return (
        f"Tiraj AMIGO:\n"
        f"Albastre (7): {blue}\n"
        f"Galbene (5): {yellow}\n\n"
        f"Rezultatul biletului tău:\n"
        f"Numere albastre găsite: {blue_hits}\n"
        f"Numere galbene găsite: {yellow_hits}\n"
    )
